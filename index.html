        function switchTab(tabId, btn) {

            document.querySelectorAll('.tab-content').forEach(el => el.classList.remove('active'));

            if(tabId === 'stats') calculateStats();

        }


        function toggleMatchFields() {

            const isMatch = document.getElementById('sessionType').value === 'match';

            document.getElementById('matchFields').style.display = isMatch ? 'block' : 'none';

        }


        function toggleModalityFields() {

            const isDoubles = document.getElementById('modality').value === 'Dobles';

        }


        function fetchWeather() {

        function saveRecord(e) {

            e.preventDefault();

            const sessionType = document.getElementById('sessionType').value;

            

            const newRecord = {

                id: Date.now(),

                sessionType,

                matchCategory: sessionType === 'match' ? document.getElementById('matchCategory').value : 'Entreno',

                modality: sessionType === 'match' ? document.getElementById('modality').value : '-',

                partner: document.getElementById('partner').value,

                rivalName: sessionType === 'match' ? document.getElementById('rivalName').value : '-',

                resultScore: document.getElementById('resultScore').value || '-',

                date: document.getElementById('date').value,

                time: document.getElementById('time').value,

                location: document.getElementById('location').value,

                weather: document.getElementById('weather').value,

                surface: document.getElementById('surface').value,

                ballType: document.getElementById('ballType').value,

                racket: document.getElementById('racket').value,

                duration: Number(document.getElementById('duration').value) || 0,

                avgHeartRate: Number(document.getElementById('avgHeartRate').value) || 0,

                calories: Number(document.getElementById('calories').value) || 0,

                notes: document.getElementById('notes').value

            };

                return;

            }

                    <p style="font-size:0.8rem; color:var(--text-muted); margin-top:4px;">⌚ Apple Watch: ${r.duration} mins | 💓 ${r.avgHeartRate} bpm | 🔥 ${r.calories} kcal | ⛅ ${r.weather || 'N/D'}</p>

            `).join('');

        }
